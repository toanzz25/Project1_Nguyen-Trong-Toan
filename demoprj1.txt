# PRO LAB SOFTWARE - HP8903B CONTROL (PRO UI - 2 COLUMN LAYOUT)

import tkinter as tk
from tkinter import ttk, filedialog, messagebox
import threading, time, csv, json
import numpy as np
import pyvisa
import matplotlib.pyplot as plt
from matplotlib.backends.backend_tkagg import FigureCanvasTkAgg

class HP8903BApp:
    def __init__(self, root):
        self.root = root
        self.root.title("HP8903B PRO LAB")
        self.root.geometry("1300x750")

        style = ttk.Style()
        style.configure("TButton", padding=6)
        style.configure("TLabel", padding=4)

        self.rm = None
        self.inst = None
        self.running = False

        self.freqs = []
        self.values = []
        self.overlays = []

        self.create_ui()

    def create_ui(self):
        main_frame = ttk.Frame(self.root)
        main_frame.pack(fill="both", expand=True)

        # LEFT (CONTROL)
        self.left_frame = ttk.Frame(main_frame, padding=15)
        self.left_frame.pack(side="left", fill="y")

        # RIGHT (GRAPH)
        self.right_frame = ttk.Frame(main_frame)
        self.right_frame.pack(side="right", fill="both", expand=True)

        self.build_control()
        self.build_graph()

    # ================= CONTROL =================
    def build_control(self):
        f = self.left_frame

        ttk.Label(f, text="Sweep Settings", font=("Segoe UI", 12, "bold")).pack(pady=10)

        grid_frame = ttk.Frame(f)
        grid_frame.pack()

        grid_frame.columnconfigure(0, weight=1)
        grid_frame.columnconfigure(1, weight=1)

        self.start = ttk.Entry(grid_frame); self.start.insert(0, "100")
        self.stop = ttk.Entry(grid_frame); self.stop.insert(0, "10000")
        self.points = ttk.Entry(grid_frame); self.points.insert(0, "50")

        ttk.Label(grid_frame, text="Start (Hz)").grid(row=0, column=0, sticky="e", padx=5, pady=5)
        self.start.grid(row=0, column=1, sticky="ew", padx=5, pady=5)

        ttk.Label(grid_frame, text="Stop (Hz)").grid(row=1, column=0, sticky="e", padx=5, pady=5)
        self.stop.grid(row=1, column=1, sticky="ew", padx=5, pady=5)

        ttk.Label(grid_frame, text="Points").grid(row=2, column=0, sticky="e", padx=5, pady=5)
        self.points.grid(row=2, column=1, sticky="ew", padx=5, pady=5)

        ttk.Label(grid_frame, text="Mode").grid(row=3, column=0, sticky="e", padx=5, pady=5)
        self.mode = ttk.Combobox(grid_frame, values=["Linear", "Log"])
        self.mode.current(1)
        self.mode.grid(row=3, column=1, sticky="ew", padx=5, pady=5)

        ttk.Label(grid_frame, text="Measurement").grid(row=4, column=0, sticky="e", padx=5, pady=5)
        self.measure = ttk.Combobox(grid_frame, values=["Voltage", "THD", "Gain"])
        self.measure.current(0)
        self.measure.grid(row=4, column=1, sticky="ew", padx=5, pady=5)

        # BUTTONS
        btn_frame = ttk.Frame(f)
        btn_frame.pack(pady=15, fill="x")

        ttk.Button(btn_frame, text="Start Sweep", command=self.start_sweep).pack(fill="x", pady=3)
        ttk.Button(btn_frame, text="Stop", command=self.stop_sweep).pack(fill="x", pady=3)
        ttk.Button(btn_frame, text="Export CSV", command=self.export_csv).pack(fill="x", pady=3)
        ttk.Button(btn_frame, text="Save Session", command=self.save_session).pack(fill="x", pady=3)
        ttk.Button(btn_frame, text="Load Session", command=self.load_session).pack(fill="x", pady=3)

    # ================= GRAPH =================
    def build_graph(self):
        self.fig, (self.ax1, self.ax2) = plt.subplots(2, 1)
        self.canvas = FigureCanvasTkAgg(self.fig, master=self.right_frame)
        self.canvas.get_tk_widget().pack(fill="both", expand=True)

        self.cursor_label = ttk.Label(self.right_frame, text="Cursor: ")
        self.cursor_label.pack()

        self.canvas.mpl_connect("motion_notify_event", self.on_cursor)

    # ================= DEVICE =================
    def measure_value(self, f):
        try:
            if self.inst:
                self.inst.write(f"FREQ {f}")
                time.sleep(0.05)
                return float(self.inst.read())
            else:
                return np.random.random()
        except:
            return np.random.random()

    # ================= SWEEP =================
    def sweep_worker(self):
        try:
            s = float(self.start.get())
            e = float(self.stop.get())
            n = int(self.points.get())

            if s <= 0:
                messagebox.showerror("Error", "Start frequency must > 0")
                self.running = False
                return

            self.freqs = []
            self.values = []

            freqs = np.logspace(np.log10(s), np.log10(e), n) if self.mode.get() == "Log" else np.linspace(s, e, n)

            for f in freqs:
                if not self.running:
                    break

                val = self.measure_value(f)
                self.freqs.append(f)
                self.values.append(val)

                self.update_plot()

            self.overlays.append((self.freqs.copy(), self.values.copy()))

        except Exception as ex:
            messagebox.showerror("Error", str(ex))

        self.running = False

    def start_sweep(self):
        if self.running:
            return
        self.running = True
        threading.Thread(target=self.sweep_worker, daemon=True).start()

    def stop_sweep(self):
        self.running = False

    # ================= PLOT =================
    def update_plot(self):
        self.ax1.clear()
        self.ax2.clear()

        for f, v in self.overlays:
            self.ax1.plot(f, 20 * np.log10(np.maximum(v, 1e-9)), alpha=0.3)

        if len(self.values) > 0:
            mag = 20 * np.log10(np.maximum(self.values, 1e-9))
            self.ax1.plot(self.freqs, mag)

        self.ax1.set_xscale("log")
        self.ax1.set_ylabel("Magnitude (dB)")

        if len(self.values) > 1:
            phase = np.gradient(self.values)
        else:
            phase = [0] * len(self.values)

        if len(self.values) > 0:
            self.ax2.plot(self.freqs, phase)

        self.ax2.set_xscale("log")
        self.ax2.set_xlabel("Frequency (Hz)")

        self.canvas.draw_idle()

    def on_cursor(self, event):
        if event.xdata:
            self.cursor_label.config(text=f"Cursor: {event.xdata:.2f} Hz")

    # ================= EXPORT =================
    def export_csv(self):
        f = filedialog.asksaveasfilename(defaultextension=".csv")
        if not f:
            return
        with open(f, 'w', newline='') as file:
            w = csv.writer(file)
            w.writerow(["Freq", "Val"])
            for i in range(len(self.freqs)):
                w.writerow([self.freqs[i], self.values[i]])

    def save_session(self):
        f = filedialog.asksaveasfilename(defaultextension=".json")
        if not f:
            return
        with open(f, 'w') as file:
            json.dump({"freqs": self.freqs, "values": self.values}, file)

    def load_session(self):
        f = filedialog.askopenfilename(filetypes=[("JSON", "*.json")])
        if not f:
            return
        with open(f) as file:
            data = json.load(file)
            self.freqs = data.get("freqs", [])
            self.values = data.get("values", [])
            self.update_plot()


if __name__ == '__main__':
    root = tk.Tk()
    app = HP8903BApp(root)
    root.mainloop()
